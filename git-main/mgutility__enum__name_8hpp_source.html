662MGUTILITY_CNSTXPR int strncmp_constexpr(const char *lhs, const char *rhs,
663 size_t count) noexcept {
664 for (size_t i = 0; i < count; ++i) {
665 // NOLINTNEXTLINE [cppcoreguidelines-pro-bounds-pointer-arithmetic]
666 if (lhs[i] != rhs[i] || lhs[i] == '\0' || rhs[i] == '\0') {
667 // NOLINTNEXTLINE [cppcoreguidelines-pro-bounds-pointer-arithmetic]
668 return static_cast<unsigned char>(lhs[i]) -
669 // NOLINTNEXTLINE [cppcoreguidelines-pro-bounds-pointer-arithmetic]
670 static_cast<unsigned char>(rhs[i]);
671 }
672 }
673 return 0;
674}

680 return i;
681 }

683 return 0;

1840 template <typename U, int Min, int Max>
1841 MGUTILITY_CNSTXPR static auto
1842 parse_chunk_text(enum_name_parse_result<U, Min, Max> &result,
1843 mgutility::string_view str, int chunk_min,
1844 int chunk_max) noexcept -> void {
1845#if defined(__clang__) || defined(__GNUC__)
1846#if defined(__clang__)
1847 auto end = str.rfind(']');
1848#elif defined(__GNUC__) && !defined(__clang__)
1849 auto end = str.rfind(';');
1850#endif
1851
1852 auto enum_names = str.substr(0, end);
1853
1854#elif defined(_MSC_VER)
1855 auto pos = str.find(',');
1856 if (pos == mgutility::string_view::npos)
1857 return;
1858
1859 ++pos;
1860
1861 auto end = str.rfind('>');
1862 auto enum_names = str.substr(pos, end - pos);
1863
1864#else
1865 return;
1866#endif
1867
1868 if (enum_names.empty()) {
1869 return;
1870 }
1871
1872 std::size_t idx = 0;
1873 std::size_t cursor = 0;
1874 const auto chunk_size = static_cast<std::size_t>(chunk_max - chunk_min);
1875 const auto range_offset = static_cast<std::size_t>(chunk_min - Min);
1876
1877 while (cursor < enum_names.size() && idx < chunk_size) {
1878 auto token_begin = cursor;
1879 auto comma = enum_names.find(',', cursor);
1880 if (comma == mgutility::string_view::npos) {
1881 break;
1882 }
1883
1884 auto token_end = comma;
1885 cursor = comma + 1;
1886
1887 while (token_begin < token_end && enum_names[token_begin] == ' ') {
1888 ++token_begin;
1889 }
1890 while (token_begin < token_end && enum_names[token_end - 1] == ' ') {
1891 --token_end;
1892 }
1893
1894 bool is_numeric_value = false;
1895#if defined(__clang__) || (defined(__GNUC__) && !defined(__clang__))
1896 is_numeric_value =
1897 token_begin < token_end && enum_names[token_begin] == '(';
1898 if (idx == 0 && !is_numeric_value) {
1899 for (auto scan = token_begin; scan < token_end; ++scan) {
1900 if (enum_names[scan] == '(' || enum_names[scan] == ')') {
1901 is_numeric_value = true;
1902 break;
1903 }
1904 }
1905 }
1906#else
1907 for (auto scan = token_begin; scan < token_end; ++scan) {
1908 if (enum_names[scan] == '(' || enum_names[scan] == ')') {
1909 is_numeric_value = true;
1910 break;
1911 }
1912 }
1913#endif
1914
1915 const auto range_index = range_offset + idx;
1916 ++idx;
1917 if (is_numeric_value) {
1918 continue;
1919 }
1920
1921 const auto scope_separator = enum_names.find(':', token_begin);
1922 if (scope_separator < token_end) {
1923 token_begin = scope_separator + 2;
1924 }
1925
1926 const auto token_size = token_end - token_begin;
1927 // Write into the fixed-size global buffer
1928 auto offset = result.strings.size();
1929 result.strings.append(enum_names.substr(token_begin, token_size));
1930#if MGUTILITY_CPLUSPLUS > 201402L
1931 result.name_lengths[range_index] = static_cast<
1932 typename enum_name_parse_result<U, Min, Max>::string_index_type>(
1933 result.strings.size() - offset);
1934#else
1935 result.ranges[range_index] = {offset, token_size};
1936#endif
1937 }
