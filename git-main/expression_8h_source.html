20// This will check if there is any way it could find a operator like member or friend and uses it.
21// If not it doesn't find the operator or if the operator at global scope is defined after
22// this template, the template won't be instantiated due to SFINAE. Once the template is not
23// instantiated it can look for global operator using normal conversions.

69template <typename L>
70struct Expression_lhs {
71 L lhs;
72 assertType::Enum m_at;
73
74 // NOLINTNEXTLINE(cppcoreguidelines-rvalue-reference-param-not-moved)
75 explicit Expression_lhs(L &&in, assertType::Enum at)
76 : lhs(static_cast<L &&>(in)), m_at(at) {}
77
78 DOCTEST_NOINLINE operator Result() {
79 // this is needed only for MSVC 2015
80 DOCTEST_MSVC_SUPPRESS_WARNING_WITH_PUSH(4800) // 'int': forcing value to bool
81 bool res = static_cast<bool>(lhs); // NOLINT(bugprone-non-zero-enum-to-bool-conversion)
82 DOCTEST_MSVC_SUPPRESS_WARNING_POP
83 if (m_at & assertType::is_false) {
84 res = !res;
85 }
86
87 if (!res || getContextOptions()->success) {
88 return {res, (DOCTEST_STRINGIFY(lhs))};
89 }
90 return {res};
91 }
92
93 /* This is required for user-defined conversions from Expression_lhs to L */
94 operator L() const {
95 return lhs;
96 }

149 // The right operator for capturing expressions is "<=" instead of "<<" (based on the operator
150 // precedence table) but then there will be warnings from GCC about "-Wparentheses" and since
151 // "_Pragma()" is problematic this will stay for now...
152 // https://github.com/catchorg/Catch2/issues/870
153 // https://github.com/catchorg/Catch2/issues/565
154 template <typename L>
155 Expression_lhs<const L &&>
156 operator<<(const L &&operand) { // bitfields bind to universal ref but not const rvalue ref
157 return Expression_lhs<const L &&>(static_cast<const L &&>(operand), m_at);
158 }
159
160 template <
161 typename L,
162 typename types::enable_if<!doctest::detail::types::is_rvalue_reference<L>::value, void>::type * = nullptr>
163 Expression_lhs<const L &> operator<<(const L &operand) {
164 return Expression_lhs<const L &>(operand, m_at);
165 }
